const fs=require('node:fs');
const {createRequire}=require('node:module');
const req=createRequire('/tmp/shady-trader-human-test/package.json');
const {chromium}=req('playwright-core');
const cfg=req('dotenv').parse(fs.readFileSync('/tmp/shady-trader-human-test/.env'));
const dir='/tmp/shady-live-review';
(async()=>{
 const browser=await chromium.launch({headless:true,executablePath:'/home/creekz/.cache/ms-playwright/chromium-1223/chrome-linux64/chrome'});
 const context=await browser.newContext({viewport:{width:1440,height:1000}});
 const page=await context.newPage();
 const record={observedAt:new Date().toISOString(),requests:[],console:[],pageErrors:[],ws:[],screens:[]};
 page.on('console',m=>{if(['error','warning'].includes(m.type()))record.console.push({type:m.type(),text:m.text().slice(0,600)});});
 page.on('pageerror',e=>record.pageErrors.push(e.message));
 page.on('response',r=>{if(r.url().includes('/api/'))record.requests.push({url:r.url(),status:r.status(),at:Date.now()});});
 page.on('requestfailed',r=>record.requests.push({url:r.url(),error:r.failure()?.errorText}));
 page.on('websocket',ws=>{record.ws.push({event:'open',url:ws.url()});ws.on('framereceived',e=>{try{const p=JSON.parse(String(e.payload));if(record.ws.length<70)record.ws.push({event:'received',type:p.type,at:Date.now(),payload:p});}catch{}});ws.on('socketerror',e=>record.ws.push({error:String(e)}));});
 await page.goto('http://127.0.0.1:3000',{waitUntil:'domcontentloaded'});
 await page.getByRole('heading',{name:'Operator Authentication'}).waitFor();
 await page.waitForTimeout(2000);
 record.preAuthRequests=record.requests.slice();
 const inputs=page.locator('input[type=password]');
 await inputs.nth(0).fill(cfg.API_ADMIN_TOKEN);
 await inputs.nth(1).fill(cfg.API_TRADER_TOKEN);
 await page.getByRole('button',{name:'Enter Dashboard'}).click();
 await page.waitForTimeout(20000);
 record.desktopText=await page.locator('body').innerText();
 record.buttons=await page.getByRole('button').allTextContents();
 await page.screenshot({path:dir+'/dashboard-desktop.png',fullPage:true});
 record.screens.push('dashboard-desktop.png');
 await page.getByRole('button',{name:'Open ML monitoring dashboard'}).click();
 await page.waitForTimeout(2000);
 record.mlText=await page.locator('body').innerText();
 await page.screenshot({path:dir+'/ml-panel.png',fullPage:true});
 await page.getByRole('button',{name:'Close ML dashboard'}).click();
 const freq=page.getByRole('button',{name:/Freqtrade/i}).first();
 if(await freq.count()){await freq.click();await page.waitForTimeout(2000);record.freqtradeText=await page.locator('body').innerText();await page.getByRole('button',{name:'Close Freqtrade panel'}).click();}
 await page.setViewportSize({width:390,height:844});
 await page.waitForTimeout(1000);
 record.mobile=await page.evaluate(()=>({viewport:innerWidth,scrollWidth:document.documentElement.scrollWidth,bodyWidth:document.body.scrollWidth}));
 await page.screenshot({path:dir+'/dashboard-mobile.png',fullPage:true});
 fs.writeFileSync(dir+'/browser.json',JSON.stringify(record,null,2));
 console.log(JSON.stringify({pageErrors:record.pageErrors,console:record.console.slice(0,12),failedResponses:record.requests.filter(r=>r.error||r.status>=400),preAuthRequests:record.preAuthRequests,mobile:record.mobile,desktopText:record.desktopText.slice(0,5500),buttons:record.buttons,wsTypes:[...new Set(record.ws.map(w=>w.type||w.event))]},null,2));
 await browser.close();
})().catch(e=>{console.error(e);process.exitCode=1});
