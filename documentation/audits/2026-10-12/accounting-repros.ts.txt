 const sqlLog:any[]=[];

 trader.portfolios.aggressive.openTrades.push({...trade});
 const baseline=trader.portfolios.aggressive.balance;
 const settlements:any[]=[];
 await trader.updatePositions(102,RiskMode.AGGRESSIVE,{recordTradeResult:async(pnl:number,principal:number)=>{settlements.push({pnl,principal});}});

 INSERT INTO balances VALUES('default',0,1000,0,0,0);

  if(sql.includes('balances')){const stmt=db.prepare(sql);return type==='all'?stmt.all(...p):stmt.run(...p);}
  return type==='all'?[]:{changes:1};
 };
 setMockRunQuery(query);setMockRunTransaction(async work=>work(query));
 const manager=new BalanceManager();
 const switched=new ShadowTrader();
 switched.portfolios.moderate.openTrades.push({...trade,id:'reserved-moderate',amount:1,leverage:1,risk_mode:RiskMode.MODERATE});
 switched.portfolios.aggressive.openTrades.push({...trade,id:'shadow-only-aggressive',amount:2,leverage:1,risk_mode:RiskMode.AGGRESSIVE});
 await manager.addActiveTrade(100);
 const reserved=await manager.getBalances();
 await switched.closeTrade('reserved-moderate',100,RiskMode.AGGRESSIVE,manager);
 const afterOldClose=await manager.getBalances();

 result.modeSwitch={fixture:'Positions staged as one initially active moderate trade with 100 reserved, plus one shadow-only aggressive trade worth200; active mode then aggressive; both close flat; actual BalanceManager with in-memory SQLite',initialWallet:1000,reserved,afterOldClose,afterBothClosed:await manager.getBalances(),expectedFinalWallet:1000};
 db.close();
 setMockRunQuery(async(sql,p,type)=>sql.includes('netPnl')?[{netPnl:0}]:sql.includes('SELECT side, amount, price')?[{side:'buy',amount:1,price:50000}]:type==='all'?[]:{changes:1});
