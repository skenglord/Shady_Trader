const fs = require('node:fs');
const {createRequire} = require('node:module');
const req = createRequire('/tmp/shady-trader-human-test/package.json');
const axios = req('axios');
const cfg = req('dotenv').parse(fs.readFileSync('/tmp/shady-trader-human-test/.env'));
const out = process.env.AUDIT_OUTPUT_DIR || '/tmp/shady-live-review';
fs.mkdirSync(out, {recursive:true});
function redact(x) {
 if (Array.isArray(x)) return x.map(redact);
 if (x && typeof x==='object') return Object.fromEntries(Object.entries(x).map(([k,v])=>[k, /secret|password|token|api.?key/i.test(k)?'[redacted]':redact(v)]));
 return x;
}
async function get(name,url,auth=false) {
 const start = Date.now();
 try {
  const r=await axios.get(url,{timeout:20000,headers:auth?{'x-api-token':cfg.API_TRADER_TOKEN}:{},validateStatus:()=>true});
  const result={name,url,observedAt:new Date().toISOString(),elapsedMs:Date.now()-start,status:r.status,headers:{date:r.headers.date,cacheControl:r.headers['cache-control'],contentType:r.headers['content-type']},body:redact(r.data)};
  fs.writeFileSync(`${out}/${name}.json`,JSON.stringify(result,null,2));
  const a=Array.isArray(r.data)?r.data:null;
  console.log(JSON.stringify({name,status:r.status,elapsedMs:result.elapsedMs,summary:a?{count:a.length,first:a[0],last:a.at(-1)}:r.data}).slice(0,2400));
 }catch(e){const result={name,url,observedAt:new Date().toISOString(),elapsedMs:Date.now()-start,error:e.message};fs.writeFileSync(`${out}/${name}.json`,JSON.stringify(result,null,2));console.log(JSON.stringify(result));}
}
