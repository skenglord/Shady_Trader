      if (this.isShuttingDown) return;
      this.isShuttingDown = true;

      // Stop main loop
      this.isRunning = false;

      if (runnerState.partialExits >= maxExits) {
        continue; // Max partial exits reached
      }
      
      // Calculate close amount as percentage of REMAINING position
      const exitFactor = config.runnerConditions?.partialExit || 0.6;
      const closeAmount = runnerState.remainingAmount * exitFactor;
      
      // Ensure we don't exit more than remaining
      const actualExitAmount = Math.min(closeAmount, runnerState.remainingAmount);
      const closeMargin = actualExitAmount * trade.price / leverage;
      const partialPnl = (currentMargin - marginUsed) * (actualExitAmount / trade.amount);
      
      portfolio.balance += partialPnl;
      trade.amount -= actualExitAmount;
      runnerState.remainingAmount -= actualExitAmount;
      runnerState.cumulativeExited += actualExitAmount;
      runnerState.partialExits += 1;
      runnerState.lastExitPrice = currentPrice;
      trade.isRunner = true;
      
      // Lock in minimum profit on remaining position
      trade.stopLoss = Math.max(trade.stopLoss, trade.price * 1.005);
      
      this.runnerStates.set(trade.id, runnerState);
      
      // ... rest of existing logic
    }

- **Runner Logic:** Edge cases in partial exits
